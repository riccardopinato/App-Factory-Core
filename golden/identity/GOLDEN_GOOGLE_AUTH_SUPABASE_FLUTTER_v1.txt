```dart
import 'package:flutter/foundation.dart';

abstract final class AuthConfig {
  static const supabaseUrl = String.fromEnvironment(
    'SUPABASE_URL',
    defaultValue: '{{SUPABASE_URL}}',
  );

  static const supabasePublishableKey = String.fromEnvironment(
    'SUPABASE_PUBLISHABLE_KEY',
    defaultValue: '{{SUPABASE_PUBLISHABLE_KEY}}',
  );

  static const webRedirect = '{{WEB_AUTH_REDIRECT}}';

  static const mobileRedirect =
      '{{MOBILE_AUTH_SCHEME}}://{{MOBILE_AUTH_HOST}}/';

  static String get oauthRedirect => kIsWeb ? webRedirect : mobileRedirect;

  static bool get configured =>
      supabaseUrl.trim().isNotEmpty &&
      supabasePublishableKey.trim().isNotEmpty;
}
```

```dart
import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import 'auth_config.dart';

enum AppAuthState {
  disabled,
  initializing,
  signedOut,
  signedIn,
  error,
}

class AppAuthService extends ChangeNotifier {
  AppAuthService._();

  static final AppAuthService instance = AppAuthService._();

  SupabaseClient? _client;
  StreamSubscription<AuthState>? _subscription;

  AppAuthState _state = AppAuthState.disabled;
  String? _lastError;
  bool _initialized = false;
  bool _initializing = false;

  AppAuthState get state => _state;
  String? get lastError => _lastError;
  bool get initialized => _initialized;
  bool get configured => AuthConfig.configured;

  User? get user => _client?.auth.currentUser;
  bool get signedIn => user != null;

  String? get userId => user?.id;
  String? get email => user?.email;

  String? get displayName =>
      user?.userMetadata?['full_name']?.toString() ??
      user?.userMetadata?['name']?.toString();

  String? get avatarUrl =>
      user?.userMetadata?['avatar_url']?.toString() ??
      user?.userMetadata?['picture']?.toString();

  String get userFacingError {
    final raw = (_lastError ?? '').toLowerCase();

    if (raw.contains('network') ||
        raw.contains('socket') ||
        raw.contains('clientexception') ||
        raw.contains('fetch')) {
      return 'Connessione non disponibile. Riprova quando torni online.';
    }

    if (raw.contains('rate limit') ||
        raw.contains('too many requests')) {
      return 'Troppi tentativi ravvicinati. Riprova tra poco.';
    }

    return 'Accesso non riuscito. Riprova.';
  }

  Future<void> initialize() async {
    if (_initialized || _initializing) return;

    if (!configured) {
      _state = AppAuthState.disabled;
      _initialized = true;
      notifyListeners();
      return;
    }

    _initializing = true;
    _state = AppAuthState.initializing;
    _lastError = null;
    notifyListeners();

    try {
      await Supabase.initialize(
        url: AuthConfig.supabaseUrl,
        publishableKey: AuthConfig.supabasePublishableKey,
      );

      _client = Supabase.instance.client;

      await _completeWebAuthCallbackIfNeeded();

      _initialized = true;
      _state = signedIn
          ? AppAuthState.signedIn
          : AppAuthState.signedOut;

      _subscription = _client!.auth.onAuthStateChange.listen((_) {
        _state = signedIn
            ? AppAuthState.signedIn
            : AppAuthState.signedOut;
        _lastError = null;
        notifyListeners();
      });
    } catch (error) {
      _client = null;
      _initialized = false;
      _state = AppAuthState.error;
      _lastError = error.toString();
    } finally {
      _initializing = false;
      notifyListeners();
    }
  }

  Future<void> signInWithGoogle() async {
    final client = _requireClient();

    _state = AppAuthState.initializing;
    _lastError = null;
    notifyListeners();

    try {
      final started = await client.auth.signInWithOAuth(
        OAuthProvider.google,
        redirectTo: AuthConfig.oauthRedirect,
        scopes: 'openid email profile',
        authScreenLaunchMode: kIsWeb
            ? LaunchMode.platformDefault
            : LaunchMode.externalApplication,
      );

      if (!started) {
        throw const AuthException(
          'Google sign-in could not be started.',
        );
      }
    } catch (error) {
      _state = AppAuthState.error;
      _lastError = error.toString();
      notifyListeners();
      rethrow;
    }
  }

  Future<void> signOut() async {
    final client = _requireClient();
    await client.auth.signOut();

    _state = AppAuthState.signedOut;
    _lastError = null;
    notifyListeners();
  }

  Future<void> _completeWebAuthCallbackIfNeeded() async {
    if (!kIsWeb) return;

    final client = _client;
    if (client == null || client.auth.currentSession != null) return;

    final code = Uri.base.queryParameters['code'];
    if (code == null || code.trim().isEmpty) return;

    try {
      await client.auth.exchangeCodeForSession(code);
    } on AuthException {
      // Non bloccare l'app: l'utente può ripetere il login.
    }
  }

  SupabaseClient _requireClient() {
    final client = _client;
    if (client == null) {
      throw StateError(
        configured
            ? 'Auth non ancora inizializzata.'
            : 'Auth non configurata.',
      );
    }
    return client;
  }

  @override
  void dispose() {
    _subscription?.cancel();
    super.dispose();
  }
}
```

6. FILE — lib/features/profile/google_auth_card.dart
======================================================================

```dart
import 'package:flutter/material.dart';

import '../../core/auth/auth_service.dart';

class GoogleAuthCard extends StatefulWidget {
  const GoogleAuthCard({super.key});

  @override
  State<GoogleAuthCard> createState() => _GoogleAuthCardState();
}

class _GoogleAuthCardState extends State<GoogleAuthCard> {
  bool busy = false;

  Future<void> _google() async {
    setState(() => busy = true);
    try {
      await AppAuthService.instance.signInWithGoogle();
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(AppAuthService.instance.userFacingError),
        ),
      );
    } finally {
      if (mounted) setState(() => busy = false);
    }
  }

  Future<void> _logout() async {
    setState(() => busy = true);
    try {
      await AppAuthService.instance.signOut();
    } finally {
      if (mounted) setState(() => busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final auth = AppAuthService.instance;

    return AnimatedBuilder(
      animation: auth,
      builder: (context, _) {
        if (!auth.configured) {
          return const ListTile(
            leading: Icon(Icons.cloud_off_outlined),
            title: Text('Account non configurato'),
          );
        }

        if (!auth.signedIn) {
          return SizedBox(
            width: double.infinity,
            child: FilledButton.icon(
              onPressed: busy ? null : _google,
              icon: busy
                  ? const SizedBox(
                      width: 18,
                      height: 18,
                      child: CircularProgressIndicator(strokeWidth: 2),
                    )
                  : const Icon(Icons.login),
              label: const Text('Continua con Google'),
            ),
          );
        }

        return ListTile(
          leading: CircleAvatar(
            backgroundImage: auth.avatarUrl == null
                ? null
                : NetworkImage(auth.avatarUrl!),
            child: auth.avatarUrl == null
                ? const Icon(Icons.person_outline)
                : null,
          ),
          title: Text(
            auth.displayName?.trim().isNotEmpty == true
                ? auth.displayName!
                : (auth.email ?? 'Account'),
          ),
          subtitle: auth.displayName?.trim().isNotEmpty == true
              ? Text(auth.email ?? '')
              : null,
          trailing: TextButton(
            onPressed: busy ? null : _logout,
            child: const Text('Esci'),
          ),
        );
      },
    );
  }
}
```

7. BOOTSTRAP APP
======================================================================

Prima di runApp o all'avvio asincrono dell'app:

```dart
await AppAuthService.instance.initialize();
```

Per app local-first:
un errore Auth NON deve impedire l'avvio del core locale.

Pattern:

```dart
try {
  await AppAuthService.instance.initialize();
} catch (_) {
  // Il core locale continua.
}
```

12. FAILURE MODES
======================================================================

OAuth non torna all'app:
- scheme/host diversi tra Dart, Manifest e Supabase.

Web loop / callback fallita:
- redirect non allow-listed;
- base URL errata;
- PKCE callback aperta in contesto incompatibile.

Login parte ma profilo vuoto:
- scope mancanti;
- metadata provider non disponibili;
- usare fallback email.

Sessione persa:
- inizializzazione duplicata;
- Supabase client ricreato;
- storage browser/device cancellato.
