======================================================================
1. SCOPO
======================================================================

Storage locale cifrato per:
- token locali non gestiti direttamente dal provider;
- chiavi applicative locali;
- credenziali create dall'utente;
- piccoli valori sensibili;
- metadati privati.

NON usare per:
- server secret;
- service-role key;
- Google client secret;
- password backend;
- segreti che non devono mai arrivare al client.

ARCHITETTURA:
- master key casuale 256 bit;
- master key protetta da Android Keystore;
- payload cifrato AES-GCM;
- nonce casuale 96 bit;
- dati cifrati salvati in SharedPreferences;
- plaintext e key buffer azzerati quando possibile;
- destroy esplicito;
- fail closed se il Keystore non riesce a decifrare.

Deriva dal pattern reale della Private Vault di Anna's Diary:
- AES-GCM;
- AndroidKeyStore;
- MethodChannel;
- FLAG_SECURE;
- zeroizzazione;
- lock lifecycle;
- test di assenza plaintext;
- recovery hardening nella v0.90.

```dart
import 'dart:convert';
import 'dart:math';
import 'dart:typed_data';

import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';
import 'package:pointycastle/export.dart';
import 'package:shared_preferences/shared_preferences.dart';

class SecureValueStore {
  SecureValueStore._();

  static final SecureValueStore instance = SecureValueStore._();

  static const _prefsKey = 'secure_value_store_v1';
  static const _wrappedKeyPrefs = 'secure_value_master_wrap_v1';

  static const _tagBits = 128;
  static const _nonceLength = 12;
  static const _masterKeyLength = 32;

  static const _aad = 'standard-secure-value-store-v1';

  static const MethodChannel _native =
      MethodChannel('standard/secure_storage');

  final Random _random = Random.secure();

  SharedPreferences? _prefs;
  Uint8List? _masterKey;
  Map<String, String> _values = const {};

  bool get unlocked => _masterKey != null;

  Future<void> initialize() async {
    _prefs ??= await SharedPreferences.getInstance();
  }

  Future<void> unlock() async {
    await initialize();

    if (kIsWeb) {
      throw UnsupportedError(
        'Persistent secure storage is disabled on Web preview.',
      );
    }

    final prefs = _prefs!;
    var wrapped = prefs.getString(_wrappedKeyPrefs);

    if (wrapped == null || wrapped.isEmpty) {
      final master = _randomBytes(_masterKeyLength);

      try {
        wrapped = await _native.invokeMethod<String>(
          'protectMasterKey',
          {'key': base64UrlEncode(master)},
        );

        if (wrapped == null || wrapped.isEmpty) {
          throw StateError('secure_master_wrap_failed');
        }

        final saved =
            await prefs.setString(_wrappedKeyPrefs, wrapped);

        if (!saved) {
          throw StateError('secure_master_wrap_persist_failed');
        }

        _masterKey = Uint8List.fromList(master);
      } finally {
        _zero(master);
      }
    } else {
      final encoded = await _native.invokeMethod<String>(
        'unprotectMasterKey',
        {'blob': wrapped},
      );

      if (encoded == null || encoded.isEmpty) {
        throw StateError('secure_master_unwrap_failed');
      }

      final master = base64Url.decode(encoded);

      try {
        if (master.length != _masterKeyLength) {
          throw StateError('secure_master_key_length_invalid');
        }
        _masterKey = Uint8List.fromList(master);
      } finally {
        _zero(master);
      }
    }

    await _load();
  }

  String? read(String key) {
    _requireUnlocked();
    return _values[key];
  }

  Future<void> write(String key, String value) async {
    _requireUnlocked();

    final cleanKey = key.trim();
    if (cleanKey.isEmpty) {
      throw const FormatException('Secure key cannot be empty.');
    }

    _values = {
      ..._values,
      cleanKey: value,
    };

    await _persist();
  }

  Future<void> delete(String key) async {
    _requireUnlocked();

    if (!_values.containsKey(key)) return;

    final next = Map<String, String>.from(_values)
      ..remove(key);

    _values = Map.unmodifiable(next);
    await _persist();
  }

  Future<void> clear() async {
    _requireUnlocked();
    _values = const {};
    await _persist();
  }

  void lock() {
    final master = _masterKey;
    if (master != null) _zero(master);

    _masterKey = null;
    _values = const {};
  }

  Future<void> destroy() async {
    await initialize();

    lock();

    await _prefs!.remove(_prefsKey);
    await _prefs!.remove(_wrappedKeyPrefs);

    if (!kIsWeb) {
      try {
        await _native.invokeMethod<void>('deleteMasterKey');
      } catch (_) {}
    }
  }

  Future<void> setSecureScreen(bool enabled) async {
    if (kIsWeb) return;

    try {
      await _native.invokeMethod<void>(
        'setSecureScreen',
        {'enabled': enabled},
      );
    } catch (_) {}
  }

  Future<void> _load() async {
    final master = _requireUnlocked();
    final raw = _prefs!.getString(_prefsKey);

    if (raw == null || raw.isEmpty) {
      _values = const {};
      return;
    }

    final envelope =
        Map<String, dynamic>.from(jsonDecode(raw) as Map);

    final plaintext = _decrypt(
      key: master,
      envelope: envelope,
      aad: _aad,
    );

    try {
      final decoded = jsonDecode(utf8.decode(plaintext));

      if (decoded is! Map) {
        throw StateError('secure_payload_invalid');
      }

      _values = Map.unmodifiable(
        decoded.map(
          (key, value) => MapEntry(
            key.toString(),
            value.toString(),
          ),
        ),
      );
    } finally {
      _zero(plaintext);
    }
  }

  Future<void> _persist() async {
    final master = _requireUnlocked();

    final plaintext = Uint8List.fromList(
      utf8.encode(jsonEncode(_values)),
    );

    try {
      final envelope = _encrypt(
        key: master,
        plaintext: plaintext,
        aad: _aad,
      );

      final saved = await _prefs!.setString(
        _prefsKey,
        jsonEncode(envelope),
      );

      if (!saved) {
        throw StateError('secure_payload_write_failed');
      }
    } finally {
      _zero(plaintext);
    }
  }

  Map<String, dynamic> _encrypt({
    required Uint8List key,
    required Uint8List plaintext,
    required String aad,
  }) {
    final nonce = _randomBytes(_nonceLength);

    final cipher = GCMBlockCipher(AESEngine())
      ..init(
        true,
        AEADParameters(
          KeyParameter(key),
          _tagBits,
          nonce,
          Uint8List.fromList(utf8.encode(aad)),
        ),
      );

    final encrypted = cipher.process(plaintext);

    return {
      'v': 1,
      'nonce': base64UrlEncode(nonce),
      'data': base64UrlEncode(encrypted),
    };
  }

  Uint8List _decrypt({
    required Uint8List key,
    required Map<String, dynamic> envelope,
    required String aad,
  }) {
    if ((envelope['v'] as num?)?.toInt() != 1) {
      throw StateError('secure_payload_version_unsupported');
    }

    final nonce =
        base64Url.decode(envelope['nonce'] as String);

    final encrypted =
        base64Url.decode(envelope['data'] as String);

    final cipher = GCMBlockCipher(AESEngine())
      ..init(
        false,
        AEADParameters(
          KeyParameter(key),
          _tagBits,
          nonce,
          Uint8List.fromList(utf8.encode(aad)),
        ),
      );

    return cipher.process(encrypted);
  }

  Uint8List _randomBytes(int length) =>
      Uint8List.fromList(
        List<int>.generate(
          length,
          (_) => _random.nextInt(256),
        ),
      );

  Uint8List _requireUnlocked() {
    final key = _masterKey;

    if (key == null) {
      throw StateError('secure_storage_locked');
    }

    return key;
  }

  void _zero(Uint8List bytes) {
    for (var i = 0; i < bytes.length; i++) {
      bytes[i] = 0;
    }
  }
}
```

5. ANDROID — SecureStorageBridge.kt
======================================================================

Creare:

android/app/src/main/kotlin/{{PACKAGE_PATH}}/SecureStorageBridge.kt

```kotlin
package {{PACKAGE_NAME}}

import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import android.view.Window
import android.view.WindowManager
import io.flutter.plugin.common.BinaryMessenger
import io.flutter.plugin.common.MethodChannel
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

class SecureStorageBridge(
    messenger: BinaryMessenger,
    private val window: Window,
) {
    companion object {
        private const val CHANNEL = "standard/secure_storage"
        private const val KEY_ALIAS = "{{PACKAGE_NAME}}.secure_storage.v1"
        private const val IV_BYTES = 12
    }

    init {
        MethodChannel(
            messenger,
            CHANNEL,
        ).setMethodCallHandler { call, result ->
            try {
                when (call.method) {
                    "protectMasterKey" -> {
                        val encoded =
                            call.argument<String>("key")
                                ?: throw IllegalArgumentException(
                                    "missing key"
                                )

                        val raw = Base64.decode(
                            encoded,
                            Base64.URL_SAFE,
                        )

                        result.success(protect(raw))
                    }

                    "unprotectMasterKey" -> {
                        val blob =
                            call.argument<String>("blob")
                                ?: throw IllegalArgumentException(
                                    "missing blob"
                                )

                        val raw = unprotect(blob)

                        result.success(
                            Base64.encodeToString(
                                raw,
                                Base64.URL_SAFE or
                                    Base64.NO_WRAP,
                            ),
                        )

                        raw.fill(0)
                    }

                    "deleteMasterKey" -> {
                        val store =
                            KeyStore.getInstance(
                                "AndroidKeyStore"
                            ).apply {
                                load(null)
                            }

                        if (store.containsAlias(KEY_ALIAS)) {
                            store.deleteEntry(KEY_ALIAS)
                        }

                        result.success(null)
                    }

                    "setSecureScreen" -> {
                        val enabled =
                            call.argument<Boolean>("enabled")
                                ?: false

                        if (enabled) {
                            window.addFlags(
                                WindowManager.LayoutParams.FLAG_SECURE
                            )
                        } else {
                            window.clearFlags(
                                WindowManager.LayoutParams.FLAG_SECURE
                            )
                        }

                        result.success(null)
                    }

                    else -> result.notImplemented()
                }
            } catch (error: Throwable) {
                result.error(
                    "secure_storage_native_error",
                    error.message ?: error.javaClass.simpleName,
                    null,
                )
            }
        }
    }

    private fun getOrCreateKey(): SecretKey {
        val store =
            KeyStore.getInstance("AndroidKeyStore").apply {
                load(null)
            }

        val existing =
            store.getKey(KEY_ALIAS, null) as? SecretKey

        if (existing != null) return existing

        val generator = KeyGenerator.getInstance(
            KeyProperties.KEY_ALGORITHM_AES,
            "AndroidKeyStore",
        )

        generator.init(
            KeyGenParameterSpec.Builder(
                KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or
                    KeyProperties.PURPOSE_DECRYPT,
            )
                .setBlockModes(
                    KeyProperties.BLOCK_MODE_GCM
                )
                .setEncryptionPaddings(
                    KeyProperties.ENCRYPTION_PADDING_NONE
                )
                .setRandomizedEncryptionRequired(true)
                .build(),
        )

        return generator.generateKey()
    }

    private fun protect(raw: ByteArray): String {
        val cipher =
            Cipher.getInstance("AES/GCM/NoPadding")

        cipher.init(
            Cipher.ENCRYPT_MODE,
            getOrCreateKey(),
        )

        val payload =
            cipher.iv + cipher.doFinal(raw)

        raw.fill(0)

        return Base64.encodeToString(
            payload,
            Base64.URL_SAFE or Base64.NO_WRAP,
        )
    }

    private fun unprotect(blob: String): ByteArray {
        val payload =
            Base64.decode(blob, Base64.URL_SAFE)

        require(payload.size > IV_BYTES) {
            "invalid secure storage envelope"
        }

        val iv =
            payload.copyOfRange(0, IV_BYTES)

        val encrypted =
            payload.copyOfRange(
                IV_BYTES,
                payload.size,
            )

        val cipher =
            Cipher.getInstance("AES/GCM/NoPadding")

        cipher.init(
            Cipher.DECRYPT_MODE,
            getOrCreateKey(),
            GCMParameterSpec(128, iv),
        )

        return cipher.doFinal(encrypted)
    }
}
```

8. LIFECYCLE
======================================================================

Per aree sensibili:
- lock su paused/inactive/hidden;
- azzerare chiave dalla RAM;
- FLAG_SECURE durante visualizzazione;
- disabilitare FLAG_SECURE in dispose.

Pattern:

```dart
@override
void didChangeAppLifecycleState(AppLifecycleState state) {
  if (state == AppLifecycleState.paused ||
      state == AppLifecycleState.inactive ||
      state == AppLifecycleState.hidden) {
    SecureValueStore.instance.lock();
  }
}
```

9. TEST CANONICI
======================================================================

[ ] Primo unlock crea master key.
[ ] Raw SharedPreferences NON contiene plaintext.
[ ] write/read restituisce il valore corretto.
[ ] lock rimuove accesso ai valori.
[ ] unlock successivo recupera valori.
[ ] destroy elimina payload e wrapped key.
[ ] delete singolo valore funziona.
[ ] tampering ciphertext -> decrypt failure.
[ ] tampering nonce -> decrypt failure.
[ ] cambio/assenza Android Keystore key -> fail closed.
[ ] FLAG_SECURE attivo nelle schermate sensibili.
[ ] background -> lock se richiesto.
[ ] nessun server secret nel client.
[ ] Web preview non persiste secret sensibili.
