5. FILE — lib/core/localization/app_language.dart
======================================================================

```dart
import 'package:flutter/widgets.dart';

enum LanguageMode {
  device,
  manual,
}

enum AppLanguage {
  english('en', 'English'),
  italian('it', 'Italiano'),
  spanish('es', 'Español'),
  french('fr', 'Français'),
  german('de', 'Deutsch'),
  portuguese('pt', 'Português');

  const AppLanguage(this.code, this.nativeLabel);

  final String code;
  final String nativeLabel;

  Locale get locale => Locale(code);

  static AppLanguage? fromCode(String? code) {
    if (code == null) return null;
    for (final value in values) {
      if (value.code == code.toLowerCase()) return value;
    }
    return null;
  }
}
```

6. FILE — lib/core/localization/locale_preference.dart
======================================================================

```dart
import 'package:flutter/widgets.dart';

import 'app_language.dart';

class LocalePreference {
  const LocalePreference({
    this.mode = LanguageMode.device,
    this.language,
  });

  final LanguageMode mode;
  final AppLanguage? language;

  Locale? get locale {
    if (mode == LanguageMode.device) return null;
    return language?.locale ?? const Locale('en');
  }

  Map<String, dynamic> toJson() => {
        'mode': mode.name,
        'language': language?.code,
      };

  factory LocalePreference.fromJson(Map<String, dynamic> json) {
    final mode = LanguageMode.values.firstWhere(
      (value) => value.name == json['mode'],
      orElse: () => LanguageMode.device,
    );

    return LocalePreference(
      mode: mode,
      language: AppLanguage.fromCode(json['language']?.toString()),
    );
  }
}
```

7. MATERIALAPP
======================================================================

```dart
import 'package:flutter/material.dart';
import 'package:flutter_localizations/flutter_localizations.dart';

import '../l10n/app_localizations.dart';

MaterialApp(
  locale: localePreference.locale,
  localizationsDelegates: const [
    AppLocalizations.delegate,
    GlobalMaterialLocalizations.delegate,
    GlobalWidgetsLocalizations.delegate,
    GlobalCupertinoLocalizations.delegate,
  ],
  supportedLocales: AppLocalizations.supportedLocales,
  // theme / darkTheme / themeMode...
);
```

Regola:
locale == null => Flutter segue automaticamente la lingua del dispositivo.

Per lingua non supportata:
Flutter usa la risoluzione delle supportedLocales; mantenere inglese come
template/default applicativo.

8. SELECTOR NEL PROFILO / IMPOSTAZIONI
======================================================================

```dart
Column(
  crossAxisAlignment: CrossAxisAlignment.start,
  children: [
    SegmentedButton<LanguageMode>(
      segments: const [
        ButtonSegment(
          value: LanguageMode.device,
          label: Text('Sistema'),
          icon: Icon(Icons.phone_android),
        ),
        ButtonSegment(
          value: LanguageMode.manual,
          label: Text('Manuale'),
          icon: Icon(Icons.language),
        ),
      ],
      selected: {localePreference.mode},
      onSelectionChanged: (values) {
        final mode = values.single;
        onChanged(
          LocalePreference(
            mode: mode,
            language: localePreference.language ??
                AppLanguage.english,
          ),
        );
      },
    ),
    if (localePreference.mode == LanguageMode.manual)
      DropdownButtonFormField<AppLanguage>(
        initialValue:
            localePreference.language ?? AppLanguage.english,
        items: AppLanguage.values
            .map(
              (language) => DropdownMenuItem(
                value: language,
                child: Text(language.nativeLabel),
              ),
            )
            .toList(growable: false),
        onChanged: (language) {
          if (language == null) return;
          onChanged(
            LocalePreference(
              mode: LanguageMode.manual,
              language: language,
            ),
          );
        },
      ),
  ],
)
```

Le label "Sistema", "Manuale" ecc. nella versione finale devono essere
anch'esse prese da AppLocalizations.
